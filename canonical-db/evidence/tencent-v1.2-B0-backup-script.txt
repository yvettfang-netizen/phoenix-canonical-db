#!/usr/bin/env bash
set -euo pipefail
umask 077
cd "$(dirname "$0")"
set -a
. ./.env
set +a
mkdir -p backups
stamp="$(date -u +%Y%m%dT%H%M%SZ)"
dump="backups/phoenix_canonical_${stamp}.dump"
test ! -e "$dump"
docker compose --env-file .env exec -T postgres \
  pg_dump -U phoenix_canonical_backup_runtime -d "$PHOENIX_DB_NAME" -Fc > "$dump"
digest="$(sha256sum "$dump" | awk '{print $1}')"
printf '{"database":"%s","created_utc":"%s","sha256":"%s","format":"pg_custom"}\n' "$PHOENIX_DB_NAME" "$stamp" "$digest" > "$dump.manifest.json"
echo "BACKUP $dump"
EXIT_CODE=0
